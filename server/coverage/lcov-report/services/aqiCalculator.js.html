// services/aqiCalculator.js — расчёт европейского индекса качества воздуха (EAQI)
 
const INDEX_LEVELS = [0, 20, 40, 60, 80, 100];
 
// Пороги концентраций для 5 загрязнителей европейского индекса (EAQI) в мкг/м³
const BREAKPOINTS = {
    pm2_5: [0, 10, 20, 25, 50, 75],
    pm10: [0, 20, 40, 50, 100, 150],
    no2: [0, 40, 90, 120, 230, 340],
    o3: [0, 50, 100, 130, 240, 380],
    so2: [0, 100, 200, 350, 500, 750]
};
 
// Линейная интерполяция под-индекса
function interpolate(c, cLow, cHigh, iLow, iHigh) {
    if (cHigh === cLow) return iLow;
    if (c <= cLow) return iLow;
    if (c >= cHigh) return iHigh;
    const index = ((c - cLow) / (cHigh - cLow)) * (iHigh - iLow) + iLow;
    return Math.round(index);
}
 
// Расчёт под-индекса загрязнителя по массиву порогов
function getSubIndex(c, breakpoints) {
    if (typeof c !== 'number' || Number.isNaN(c) || c <= 0) {
        return 0;
    }
 
    const intervalsCount = breakpoints.length - 1;
    for (let i = 0; i < intervalsCount; i += 1) {
        const cLow = breakpoints[i];
        const cHigh = breakpoints[i + 1];
        const iLow = INDEX_LEVELS[i];
        const iHigh = INDEX_LEVELS[i + 1];
 
        if (c <= cHigh) {
            return interpolate(c, cLow, cHigh, iLow, iHigh);
        }
    }
 
    // Экстраполяция по последнему отрезку с лимитом 150
    const lastIdx = intervalsCount;
    const cLow = breakpoints[lastIdx - 1];
    const cHigh = breakpoints[lastIdx];
    const iLow = INDEX_LEVELS[lastIdx - 1];
    const iHigh = INDEX_LEVELS[lastIdx];
 
    const extrapolated = ((c - cLow) / (cHigh - cLow)) * (iHigh - iLow) + iLow;
    return Math.min(150, Math.round(extrapolated));
}
 
// Общий индекс часа: максимум из всех под-индексов (наихудший показатель).
// Поддерживает как позиционный вызов calculateEAQI(pm25, pm10, no2, o3, so2),
// так и передачу объекта параметров calculateEAQI({ pm10, pm2_5, no2, o3, so2 }).
function calculateEAQI(arg1, pm10Arg, no2Arg, o3Arg, so2Arg) {
    let pm25, pm10, no2, o3, so2;
 
    if (typeof arg1 === 'object' && arg1 !== null) {
        pm25 = Number(arg1.pm2_5 ?? arg1.pm25 ?? 0);
        pm10 = Number(arg1.pm10 ?? 0);
        no2 = Number(arg1.no2 ?? 0);
        o3 = Number(arg1.o3 ?? 0);
        so2 = Number(arg1.so2 ?? 0);
    } else {
        pm25 = Number(arg1 ?? 0);
        pm10 = Number(pm10Arg ?? 0);
        no2 = Number(no2Arg ?? 0);
        o3 = Number(o3Arg ?? 0);
        so2 = Number(so2Arg ?? 0);
    }
 
    const subPm25 = getSubIndex(pm25, BREAKPOINTS.pm2_5);
    const subPm10 = getSubIndex(pm10, BREAKPOINTS.pm10);
    const subNo2 = getSubIndex(no2, BREAKPOINTS.no2);
    const subO3 = getSubIndex(o3, BREAKPOINTS.o3);
    const subSo2 = getSubIndex(so2, BREAKPOINTS.so2);
 
    return Math.max(subPm25, subPm10, subNo2, subO3, subSo2);
}
 
module.exports = {
    BREAKPOINTS,
    interpolate,
    getSubIndex,
    calculateEAQI
};
